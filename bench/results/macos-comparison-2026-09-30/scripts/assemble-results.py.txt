from pathlib import Path
from collections import Counter
import copy, hashlib, json, shutil, statistics, sys
sys.path.insert(0,'bench')
from macos_bench import report
base=Path('/tmp/loom-four-way-final-20260930')
kitty=Path('/tmp/loom-kitty-final-retry-20260930')
ghostty=Path('/tmp/loom-ghostty-final-20260930')
batches={label:json.loads((p/'results.json').read_text()) for label,p in [('initial',base),('kitty-retry',kitty),('ghostty-direct',ghostty)]}
d=copy.deepcopy(batches['initial'])
d['source_batches']={label:{k:v for k,v in data.items() if k!='runs'} for label,data in batches.items()}
d['runs']=[]
for label,path in [('initial',base),('kitty-retry',kitty),('ghostty-direct',ghostty)]:
    for original in batches[label]['runs']:
        r=copy.deepcopy(original)
        r['source_batch']=label
        r['source_directory']=r.get('source_directory',str(path/f"r{r['round']}-{r['app']}"))
        if label=='kitty-retry':
            r['source_round']=r['round']
            r['round']=1
        d['runs'].append(r)
valid=[r for r in d['runs'] if 'error' not in r]
assert Counter(r['app'] for r in valid)==dict.fromkeys(d['settings']['apps'],3)
assert all(len(r['phases'])==20 and (r['ready']['columns'],r['ready']['rows'])==(100,30) for r in valid)
for r in valid:
    log=(Path(r['source_directory'])/'terminal.log').read_text()
    assert 'panicked at' not in log
    if r['app']=='loomtty':
        assert r['display']==dict(cell_width=8.0,cell_height=16.0,dpi_scale=1.0)
        assert 'DPI changed:' not in log and 'config reloaded' not in log
for name in ['ascii','truecolor','cjk','emoji','scroll','cursor']:
    ps=[next(p for p in r['phases'] if p['name']==name and not p['warmup']) for r in valid]
    assert len({(p['bytes'],p['sha256']) for p in ps})==1
for label,batch in batches.items():
    for app,version in batch['versions'].items():
        assert version==d['versions'][app]
    assert batch['benchmark_sha256']['terminal_workload.py']==d['benchmark_sha256']['terminal_workload.py']
out=Path('bench/results/macos-comparison-2026-09-30')
out.mkdir(parents=True,exist_ok=True)
(out/'results.json').write_text(json.dumps(d,ensure_ascii=False,indent=2)+'\n')
report(d,out)
(out/'logs').mkdir(exist_ok=True)
for i,r in enumerate(d['runs']):
    shutil.copy2(Path(r['source_directory'])/'terminal.log',out/'logs'/f"{i:02d}-{r['app']}-r{r['round']}.log")
(out/'scripts').mkdir(exist_ok=True)
for script in ['macos_bench.py','terminal_workload.py']:
    shutil.copy2(Path('bench')/script,out/'scripts'/(script+'.txt'))
shutil.copy2('bench/results/macos-performance-2026-09-30/scripts/macos_bench.py.txt',out/'scripts'/'macos_bench.initial.py.txt')
shutil.copy2('/tmp/loom-resume-benchmark.py',out/'scripts'/'resume-controller.py.txt')
shutil.copy2(__file__,out/'scripts'/'assemble-results.py.txt')
runs={app:[r for r in valid if r['app']==app] for app in d['settings']['apps']}
def vals(app,name,key,resource=False):
    ps=[next(p for p in r['phases'] if p['name']==name and not p['warmup']) for r in runs[app]]
    return [(p['resources'] if resource else p)[key] for p in ps]
def metric(label,name,key,resource=False,digits=1):
    return '| '+label+' | '+' | '.join(f'{statistics.median(vals(app,name,key,resource)):.{digits}f}' for app in runs)+' |'

    lines.append(metric(name,name,'mib_per_second'))
